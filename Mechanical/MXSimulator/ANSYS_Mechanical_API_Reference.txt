    force = analysis.AddForce()

    # 방향 설정
    force.DefineBy = Ansys.Mechanical.DataModel.Enums.LoadDefineBy.Components  # 성분별
    # 또는
    force.DefineBy = Ansys.Mechanical.DataModel.Enums.LoadDefineBy.Vector       # 벡터

    # 시간-하중 타뷸러 데이터 (⭐ 핵심 패턴!)
    # Inputs[0] = 시간, Output = 하중값
    force.XComponent.Inputs[0].DiscreteValues = [Quantity("0 [s]"), Quantity("1 [s]")]
    force.XComponent.Output.DiscreteValues = [Quantity("100 [N]"), Quantity("200 [N]")]
    # Y축, Z축도 동일
    force.YComponent.Inputs[0].DiscreteValues = [...]
    force.YComponent.Output.DiscreteValues = [...]
    force.ZComponent.Inputs[0].DiscreteValues = [...]
    force.ZComponent.Output.DiscreteValues = [...]

    # 벡터 방식일 때:
    force.Magnitude.Inputs[0].DiscreteValues = [Quantity("0 [s]"), ...]
    force.Magnitude.Output.DiscreteValues = [Quantity("100 [N]"), ...]
    force.Direction = ...  # 방향 벡터

  [속성 목록]
    force.DefineBy           - 정의 방법 (Components/Vector)
    force.XComponent         - X 성분
    force.YComponent         - Y 성분
    force.ZComponent         - Z 성분
    force.Magnitude          - 크기 (벡터 방식)
    force.Direction          - 방향
    force.Location           - 적용 위치 (Named Selection)
    force.CoordinateSystem   - 좌표계
    force.DivideLoadByNodes  - 절점당 분배

7. SelectionManager API
================================================================================

  sel_mgr = ExtAPI.SelectionManager

  # SelectionInfo 생성
  selection = sel_mgr.CreateSelectionInfo(SelectionTypeEnum.GeometryEntities)
  selection = sel_mgr.CreateSelectionInfo(SelectionTypeEnum.MeshNodes)
  selection = sel_mgr.CreateSelectionInfo(SelectionTypeEnum.MeshElements)

  # 엔티티 설정 (공식 패턴)
  selection.Entities = [face1, face2, face3]    # ← 리스트로 직접 대입
  selection.Ids = [id1, id2, id3]               # ID로도 가능
  selection.Id = single_id                       # 단일 ID

  # 선택 관리
  sel_mgr.AddSelection(selection)               # 기존 선택에 추가
  sel_mgr.NewSelection(selection)               # 선택 교체
  sel_mgr.ClearSelection()                      # 선택 해제
  sel_mgr.CurrentSelection                      # 현재 선택 정보

  # SelectionTypeEnum 값들 (Ansys.ACT.Interfaces.Common)
    GeometryEntities    - 형상 엔티티 (Face, Edge, Vertex, Body)
    MeshNodes           - 메시 절점
    MeshElements        - 메시 요소

8. Geometry API - 바디/면/엣지/꼭짓점
================================================================================

  [바디 가져오기]
    # 방법 1: GetChildren (권장)
    from Ansys.Mechanical.DataModel.Enums import DataModelObjectCategory
    bodies = model.Geometry.GetChildren(DataModelObjectCategory.Body, True)

    # 방법 2: Geometry.Children 순회
    for part in model.Geometry.Children:
        for body in part.Children:
            pass

    # 방법 3: GeoData (IGeoAssembly)
    geo = ExtAPI.DataModel.GeoData
    for part in geo.Parts:
        for body in part.Bodies:
            pass

  [Body → GeoBody → Faces]
    body = model.Geometry.Children[0].Children[0]   # ACT Body 객체
    geo_body = body.GetGeoBody()                     # IBaseGeoBody 반환  ✅

    # IBaseGeoBody 속성
    geo_body.Faces       - 면 컬렉션 (IList of IBaseGeoFace)
    geo_body.Edges       - 엣지 컬렉션
    geo_body.Vertices    - 꼭짓점 컬렉션
    geo_body.Centroid    - 무게중심
    geo_body.Volume      - 부피
    geo_body.Area        - 면적
    geo_body.BodyType    - 바디 유형

  [IBaseGeoFace 메서드 - 면 분석]
    face = geo_body.Faces[i]

    # ⚠️ 법선 계산 (GetFaceNormal 아님!)
    normal = face.NormalAtParam(0.5, 0.5)            # u=0.5, v=0.5 (파라미터 중심)
    normals = face.NormalsAtParams([u1,v1, u2,v2])   # 여러 점의 법선 한번에

    # 점 좌표
    point = face.PointAtParam(0.5, 0.5)              # 파라미터 → 3D 좌표
    points = face.PointsAtParams([u1,v1, u2,v2])     # 여러 점

    # 파라미터
    params = face.ParamAtPoint([x, y, z])            # 3D 좌표 → 파라미터

    # 속성
    face.Area             - 면적
    face.SurfaceType      - 면 유형 (Plane, Cylinder, Cone, Sphere, ...)
    face.Centroid         - 면 무게중심
    face.Bodies           - 이 면에 속한 바디들
    face.Edges            - 이 면에 속한 엣지들
    face.Vertices         - 이 면에 속한 꼭짓점들

    # NormalAtParam 반환값 처리
    normal = face.NormalAtParam(0.5, 0.5)
    nx, ny, nz = normal[0], normal[1], normal[2]     # 인덱스 접근 (배열 형태)

  [IBaseGeoEdge 속성]
    edge.Length         - 길이
    edge.CurveType      - 곡선 유형 (Line, Arc, ...)
    edge.StartVertex    - 시작 꼭짓점
    edge.EndVertex      - 끝 꼭짓점
    edge.Centroid       - 무게중심
    edge.Faces          - 인접한 면들
    edge.Bodies         - 속한 바디들

  [IBaseGeoVertex 속성]
    vertex.X, vertex.Y, vertex.Z    - 좌표
    vertex.Edges                    - 연결된 엣지들
    vertex.Faces                    - 인접한 면들
    vertex.Bodies                   - 속한 바디들

  [IGeoAssembly (ExtAPI.DataModel.GeoData)]
    geo.Name       - 어셈블리 이름
    geo.Parts      - 파트 목록
    geo.AllParts   - 모든 파트 (숨김 포함)
    geo.Unit       - 단위
    geo.Dimension  - 차원 (2D/3D)

13. ANSYSAnalysisSettings - Per-Step 메서드 전체 목록
================================================================================

  모든 메서드는 (step_index: UInt32, value) 형태.
  step_index는 1부터 시작!

  Get/Set 메서드:
    GetStepEndTime(i)          / SetStepEndTime(i, Quantity)
